func add(a int, b int) int {
    return a + b;
};

func subtract(a int, b int) int {
    return a - b;
};

func multiply(a int, b int) int {
    return a * b;
};

func divide(a float64, b float64) float64 {
    if b == 0.0 {
        println("Division by zero");
        return 0.0;
    };

    return a / b;
};

func factorial(n int) int {
    if n <= 1 {
        return 1;
    };

    return n * factorial(n - 1);
};

func fibonacci(n int) int {
    if n <= 1 {
        return n;
    };

    return fibonacci(n - 1) + fibonacci(n - 2);
};

func printHeader() {
    println("================================");
    println(appName);
    println("Version:", version);
    println("================================");
};

func processUser(age int, score float64) string {
    if age >= 18 && score >= 70.0 {
        return "Approved";
    } else if age >= 18 && score >= 50.0 {
        return "Pending";
    } else {
        return "Rejected";
    };
};

func calculateStats(a int, b int, c int) int {
    var result int = 0;

    result += a;
    result += b;
    result += c;

    result *= 2;
    result /= 3;
    result %= 5;

    return result;
};

func testLoops() {
    var i int = 0;
    var sum int = 0;

    for i < 10 {
        sum += i;
        i++;
    };

    println("While-style loop sum:", sum);

    for j := 0; j < 5; j++ {
        println("Classic loop:", i);
    };

    for ;; {
        println("Infinite loop simulation");
        break;
    };
};

func testConditions(value int) {
    if value > 100 {
        println("Greater than 100");
    } else if value > 50 {
        println("Greater than 50");
    } else if value > 10 {
        println("Greater than 10");
    } else {
        println("10 or less");
    };
};

func testSwitch(day int) {
    switch day {
        case 1:
            println("Monday");

        case 2:
            println("Tuesday");

        case 3:
            println("Wednesday");

        case 4:
            println("Thursday");

        case 5:
            println("Friday");

        default:
            println("Weekend");
    };
};

func testMath() {
    var a int = 10;
    var b int = 3;

    println("Addition:", a + b);
    println("Subtraction:", a - b);
    println("Multiplication:", a * b);
    println("Division:", a / b);
    println("Modulo:", a % b);

    println("Equal:", a == b);
    println("Not Equal:", a != b);
    println("Greater:", a > b);
    println("Less:", a < b);

    println("Logical AND:", a > 5 && b < 5);
    println("Logical OR:", a > 100 || b < 5);

    println("Bitwise AND:", a & b);
    println("Bitwise OR:", a | b);
    println("Bitwise XOR:", a ^ b);

    println("Left Shift:", a << 1);
    println("Right Shift:", a >> 1);
};

func testAssignments() {
    var x int = 10;

    x += 5;
    println("+= :", x);

    x -= 2;
    println("-= :", x);

    x *= 3;
    println("*= :", x);

    x /= 2;
    println("/= :", x);

    x %= 4;
    println("%= :", x);

    x |= 1;
    println("|= :", x);

    x &= 7;
    println("&= :", x);

    x ^= 2;
    println("^= :", x);

    x <<= 1;
    println("<<= :", x);

    x >>= 1;
    println(">>= :", x);
};
